X_hba1c   = X_merged_diabetes[["LBXGH"]].values
M_ckd     = y_merged_ckd.values
Y_cardio  = y_merged_cardio.values

# split patients into high vs low HbA1c at the median 
hba1c_median = np.median(X_hba1c)
high_hba1c   = X_hba1c >= hba1c_median  # above median
low_hba1c    = X_hba1c <  hba1c_median  # below median

# difference in predicted cardio risk: high vs low HbA1c
hba1c_to_cardio = nn_prob(X_hba1c, Y_cardio, X_hba1c)
total_effect    = hba1c_to_cardio[high_hba1c.flatten()].mean() - \
                  hba1c_to_cardio[low_hba1c.flatten()].mean()
print(f"Total Effect  (HbA1c → Cardiovascular Risk): {total_effect}")

# difference in CKD probability: high vs low HbA1c
hba1c_to_ckd_prob  = nn_prob(X_hba1c, M_ckd, X_hba1c)
diabetes_to_kidney = hba1c_to_ckd_prob[high_hba1c.flatten()].mean() - \
                     hba1c_to_ckd_prob[low_hba1c.flatten()].mean()
print(f"Diabetes → Kidney (HbA1c → CKD):            {diabetes_to_kidney}")

# difference in cardio risk: CKD positive vs negative
XM_combined        = np.column_stack([X_hba1c, hba1c_to_ckd_prob])
combined_pred      = nn_prob(XM_combined, Y_cardio, XM_combined)
kidney_to_cardio   = combined_pred[M_ckd == 1].mean() - \
                     combined_pred[M_ckd == 0].mean()
print(f"Kidney → Cardio (CKD → CVD | HbA1c):        {kidney_to_cardio}")

# ── Direct Effect ──────────────────────────────────────────
# HbA1c effect after holding CKD constant at its mean
XM_ckd_held        = XM_combined.copy()
XM_ckd_held[:, 1]  = hba1c_to_ckd_prob.mean()
direct_pred        = nn_prob(XM_combined, Y_cardio, XM_ckd_held)
direct_effect      = direct_pred[high_hba1c.flatten()].mean() - \
                     direct_pred[low_hba1c.flatten()].mean()
print(f"Direct Effect (HbA1c → CVD controlling CKD): {direct_effect}")

# ── Indirect Effect ────────────────────────────────────────
indirect_effect     = diabetes_to_kidney * kidney_to_cardio
proportion_mediated = indirect_effect / total_effect 

print(f"\nIndirect Effect (through CKD):  {indirect_effect}")
print(f"Proportion mediated by CKD:     {proportion_mediated}")

print("\n=== MEDIATION SUMMARY ===")
hba1c_summary = pd.DataFrame([
    {"Pathway": "HbA1c → Cardiovascular Risk (total)",
     "Value":   round(total_effect, 4),
     "Meaning": "Cardio risk difference: high vs low HbA1c"},
    {"Pathway": "HbA1c → CKD (diabetes to kidney)",

     "Meaning": "% of HbA1c-cardio relationship explained by CKD"},
])
print(hba1c_summary.to_string(index=False))

fig, ax = plt.subplots(figsize=(10, 5))

# create groups by HbA1c tertile AND CKD status
low_q  = merged["LBXGH"].quantile(0.33)
high_q = merged["LBXGH"].quantile(0.66)

groups = {
    "Low HbA1c\nNo CKD":    merged[(merged["LBXGH"] <  low_q)  & (merged["ckd_label"]==0)],
    "Low HbA1c\nWith CKD":  merged[(merged["LBXGH"] <  low_q)  & (merged["ckd_label"]==1)],
    "High HbA1c\nNo CKD":   merged[(merged["LBXGH"] >= high_q) & (merged["ckd_label"]==0)],
    "High HbA1c\nWith CKD": merged[(merged["LBXGH"] >= high_q) & (merged["ckd_label"]==1)],
}

labels  = list(groups.keys())
rates   = [g["cardio_label"].mean() for g in groups.values()]
sizes   = [len(g) for g in groups.values()]
colors  = ["#aec6e8", "#d9534f", "#185FA5", "#8B0000"]

bars = ax.bar(labels, rates, color=colors, alpha=0.85, width=0.5)

for bar, val, n in zip(bars, rates, sizes):
    ax.text(bar.get_x() + bar.get_width()/2,
            bar.get_height() + 0.008,
            f'{val:.1%}', ha='center', fontsize=11, fontweight='bold')
    ax.text(bar.get_x() + bar.get_width()/2,
            bar.get_height() + 0.03,
            f'n={n}', ha='center', fontsize=8, color='gray')

ax.set_ylabel("Cardiovascular Risk Rate")
