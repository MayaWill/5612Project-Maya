# Direct Effect
# Does CKD predict cardiovascular risk controlling for diabetes?

# combine diabetes and CKD as one / jointinput
XM = np.column_stack([X_dia_pred, M_pred_prob])

Y_pred_with_M = nn_prob(XM, Y_CVD_outcome, XM)

# direct effect,  diabetes effect when CKD is held constant
# set CKD probability to its mean (counterfactual: no mediation)
XM_no_mediator       = XM.copy()
XM_no_mediator[:, 1] = M_pred_prob.mean()  # hold CKD constant

Y_pred_no_M = nn_prob(XM, Y_CVD_outcome, XM_no_mediator)

direct_effect = (
    Y_pred_no_M[X_dia_pred.flatten() == 1].mean() -
    Y_pred_no_M[X_dia_pred.flatten() == 0].mean()
)

#  CKD effect on cardio controlling for diabetes
ckd_to_cardio_dia_control = (
    Y_pred_with_M[M_ckd_med == 1].mean() -
    Y_pred_with_M[M_ckd_med == 0].mean()
)
