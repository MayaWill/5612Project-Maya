features_kidney = [
    "LBXSCR",   # serum creatinine
    "LBXSBU",   # BUN
    "LBXSAL",   # serum albumin
    "LBXSPH",   # phosphorus
    "LBXSKSI",  # potassium
    "URXUMA",   # urine albumin
    "URXUCR",   # urine creatinine
    "URDACT",   # albumin-to-creatinine ratio
]

features_diabetes = [
    "LBXGH",     # HbA1c
  #  "LBDGLUSI",  # fasting glucose
    "LBDHDD",    # HDL
    "LBXIN",     # insulin
    "LBXTC",     # total cholesterol
    "LBDLDL",    # LDL 
    "LBXTR",     # triglycerides
]


nhanes_cardio_df["SBP_AVG"] = nhanes_cardio_df[
    ["BPXOSY1", "BPXOSY2", "BPXOSY3"]
].mean(axis=1)

nhanes_cardio_df["DBP_AVG"] = nhanes_cardio_df[
    ["BPXODI1", "BPXODI2", "BPXODI3"]
].mean(axis=1)

nhanes_cardio_df["cardio_label"] = (
    (nhanes_cardio_df["LBDLDL"]   >= 130) |
    (nhanes_cardio_df["LBXTC"]    >= 240) |
    (nhanes_cardio_df["SBP_AVG"]  >= 140) |
    (nhanes_cardio_df["LBXHSCRP"] >= 3)
).astype(int)

features_cardio = [
    "LBDHDD",    # HDL cholesterol
    "LBXTR",     # triglycerides
    "DBP_AVG",   # avg diastolic BP
    "LBXHSCRP"
]
