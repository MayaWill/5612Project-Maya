#initialize model for nhanes dataset - examie 3 different k values

fig, axes = plt.subplots(1, 3, figsize=(15, 4))
for ax, (name, x_scaled, k) in zip(axes, [
    ("NHANES", x_nhanes_scaled, 2),
    ("Pima",   x_pima_scaled,   2),
    ("UCI",    x_uci_scaled,    2),
]):
    labels = KMeans(n_clusters=k, random_state=42, n_init=10).fit_predict(x_scaled)
    dim = PCA(n_components=2).fit_transform(x_scaled)
    ax.scatter(dim[:, 0], dim[:, 1], c=labels, cmap='coolwarm', s=6)
    ax.set_title(f"{name} (k={k})")
    ax.set_xlabel("PC1"); ax.set_ylabel("PC2")
plt.tight_layout()
plt.show()
