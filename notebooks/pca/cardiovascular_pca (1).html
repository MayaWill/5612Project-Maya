#initialize model for nhanes dataset - examie 3 different k values

fig, axes = plt.subplots(1, 3, figsize=(15, 4))
for ax, (name, x_scaled, k) in zip(axes, [
    ("Train", x_train_scaled, 2),
    ("Cleveland",   x_cleveland_scaled,   2),
    ("Framingham",    x_framingham_scaled,    2),
]):
    labels = KMeans(n_clusters=k, random_state=42, n_init=10).fit_predict(x_scaled)
    dim = PCA(n_components=2).fit_transform(x_scaled)
    ax.scatter(dim[:, 0], dim[:, 1], c=labels, cmap='coolwarm', s=6)
    ax.set_title(f"{name} (k={k})")
    ax.set_xlabel("PC1"); ax.set_ylabel("PC2")
plt.tight_layout()
plt.show()

print("PC Variance for x_train_scaled:")

pca = PCA(n_components=2).fit(x_train_scaled)
loadings = pd.DataFrame(pca.components_.T, index=x_train.columns, columns=['PC1','PC2'])
print(loadings.round(2))
print(pca.explained_variance_ratio_.round(2))   # how much variance each axis captures

eigenvalues = pca.explained_variance_          # eigenvalues
var_ratio   = pca.explained_variance_ratio_    # % variance per component

print("Eigenvalues:", eigenvalues.round(2))
print("Variance ratio:", var_ratio.round(2))
print("Cumulative:", var_ratio.cumsum().round(2))
