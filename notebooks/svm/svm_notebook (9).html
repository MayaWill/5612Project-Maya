def run_svm(X, y, kernel, C = 1.0):
    X_train, X_test, y_train, y_test = train_test_split(
        X,y, test_size=0.4, random_state=42, stratify=y
    )

    scaler = StandardScaler()

    X_train = scaler.fit_transform(X_train)
    X_test = scaler.transform(X_test)

    svm = SVC(kernel = kernel, C=C)

    svm.fit(X_train, y_train)

    y_pred = svm.predict(X_test)

    return {
        "kernel": kernel,
        "C": C,
        "accuracy": accuracy_score(y_test, y_pred),
        "precision": precision_score(y_test, y_pred),
        "recall": recall_score(y_test, y_pred),
        "cm": confusion_matrix(y_test, y_pred),
        "y_test": y_test,
        "y_pred": y_pred,
        "f1 scoe": f1_score(y_test, y_pred),
    }

from sklearn.metrics import ConfusionMatrixDisplay

for dataset_name, df in all_results.items():
    best = df.loc[df.groupby("kernel")["accuracy"].idxmax()]
    fig, axes = plt.subplots(1, 3, figsize=(12, 4))
    fig.suptitle(f"{dataset_name.upper()} — Best Confusion Matrix per Kernel")
    for ax, (_, row) in zip(axes, best.iterrows()):
        ConfusionMatrixDisplay.from_predictions(row["y_test"], row["y_pred"], ax=ax)
        ax.set_title(f"{row['kernel']}, C={row['C']}")
    plt.tight_layout()
    plt.show()
