cosine_sim = cosine_similarity(x_marker_scaled)
cosine_dist = 1 - cosine_sim
cosine_dist = np.clip(cosine_dist, 0, 2)  #remove negative distances
cosine_dist_condensed = squareform(cosine_dist, checks=False)

linked = linkage(cosine_dist_condensed, method='average')

plt.figure(figsize=(14, 6))
dendrogram(
    linked,
    labels=pivot.index.tolist(),
    leaf_rotation=90,
    leaf_font_size=7
)

    plt.scatter(components_jittered[i, 0], components_jittered[i, 1],
                color=colors[h_labels[i]], s=60, alpha=0.7)

for i, name in enumerate(pivot.index):
    if name in shared:
        plt.annotate(
            name,
            (components_jittered[i, 0], components_jittered[i, 1]),
            fontsize=8, fontweight='bold',
            xytext=(components_jittered[i, 0] + 0.3, components_jittered[i, 1] + 0.2),
            arrowprops=dict(arrowstyle='-', color='gray', lw=0.5)
        )
