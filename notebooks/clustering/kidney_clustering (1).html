# find best k value for each using elbow, then silhouette
# elbow and silouhette for nhanes scaled

#elbow
inertias_nhanes = []
K = range(1, 11)

for k in K:
    kmeans = KMeans(n_clusters = k, random_state=42, n_init=10)
    kmeans.fit(x_nhanes_scaled)
    inertias_nhanes.append(kmeans.inertia_)
#elbow plot for nhanes
plt.plot(K, inertias_nhanes, marker='o')
plt.title('Elbow Method for NHANES Kidney Dataset')
plt.xlabel('Number of clusters (k)')
plt.ylabel('Inertia')
plt.show()

#silhouette

K = range(2, 11)
scores_nhanes = []

for k in K:
    kmeans = KMeans(n_clusters = k, random_state = 42, n_init = 10)
    nhanes_labels = kmeans.fit_predict(x_nhanes_scaled)
    score = silhouette_score(x_nhanes_scaled, nhanes_labels)
    scores_nhanes.append(score)

plt.plot(K, scores_nhanes, marker='o')
plt.title("Silhouette Score vs K for NHANES Kidney Dataset")
plt.xlabel("Number of clusters (k)")
plt.ylabel("Silhouette Score")
plt.show()

#initialize model for nhanes dataset - examie 3 different k values

fig, axes = plt.subplots(1, 2, figsize=(15, 4))
for ax, (name, x_scaled, k) in zip(axes, [
    ("NHANES", x_nhanes_scaled, 2),
    ("UCI",    x_uci_scaled,    2),
]):
    labels = KMeans(n_clusters=k, random_state=42, n_init=10).fit_predict(x_scaled)
    dim = PCA(n_components=2).fit_transform(x_scaled)
    ax.scatter(dim[:, 0], dim[:, 1], c=labels, cmap='coolwarm', s=6)
    ax.set_title(f"{name} (k={k})")
    ax.set_xlabel("PC1"); ax.set_ylabel("PC2")
plt.tight_layout()
plt.show()
