X_copy_k = X_kidney.copy()
y_copy_k = y_kidney.copy()

# fixed split for fair comparison
X_train_ki, X_test_ki, y_train_ki, y_test_ki = train_test_split(
    X_copy_k, y_copy_k,
    test_size=0.4,
    random_state=42,
    stratify=y_copy_k
)

# bins to test
bin_values = [3, 4, 5, 6, 7, 8, 10, 12, 15, 20]

precisions = []
recalls = []

for b in bin_values:
    kb_ki = KBinsDiscretizer(n_bins=b, encode="ordinal", strategy="quantile")

    X_train_b_ki = kb_ki.fit_transform(X_train_ki)
    X_test_b_ki = kb_ki.transform(X_test_ki)

    model = MultinomialNB()
    model.fit(X_train_b_ki, y_train_ki)

    y_pred_ki = model.predict(X_test_b_ki)

    precisions.append(precision_score(y_test_ki, y_pred_ki, pos_label=1, zero_division=0))
    recalls.append(recall_score(y_test_ki, y_pred_ki, pos_label=1,zero_division=0))


#bins plot
plt.figure(figsize=(10,6))

plt.plot(bin_values, precisions, marker='o', label="Precision (Kidney)")
plt.plot(bin_values, recalls, marker='o', label="Recall (Kidney)")

# highlight chosen bin
plt.axvline(x=10, linestyle="--", color="red", label="Chosen: 10 bins")

plt.title("Effect of Binning on Multinomial Naive Bayes (NHANES Kidney)")
plt.xlabel("Number of Bins")
plt.ylabel("Score (Class 1)")
plt.ylim(0, 1)
plt.xticks(bin_values)
plt.grid(True)
plt.legend()
